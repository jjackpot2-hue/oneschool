import sys import requests from PyQt5.QtWidgets import ( QApplication, QWidget, QVBoxLayout, QHBoxLayout, QLabel, QLineEdit, QPushButton, QTableWidget, QTableWidgetItem, QMessageBox, QTabWidget ) class OnSchoolApp(QWidget): def __init__(self): super().__init__() # 나이스 인증키 (테스트용 공공 API Key 또는 발급받은 KEY 입력) self.api_key = "sample" # 필요 시 발급받은 나이스 API 키로 변경하세요 self.initUI() def initUI(self): self.setWindowTitle('온스쿨 시간표 & 급식 조회 어플') self.resize(600, 500) main_layout = QVBoxLayout() # 1. 학교 검색 영역 search_layout = QHBoxLayout() self.school_input = QLineEdit() self.school_input.setPlaceholderText("학교명을 입력하세요 (예: 초등학교, 중학교)") self.search_btn = QPushButton("학교 검색") self.search_btn.clicked.connect(self.search_school) search_layout.addWidget(self.school_input) search_layout.addWidget(self.search_btn) main_layout.addLayout(search_layout) # 학교 선택 정보 레이블 self.info_label = QLabel("학교를 검색하여 선택해주세요.") main_layout.addWidget(self.info_label) # 탭 구성 (시간표 / 급식) self.tabs = QTabWidget() # 시간표 탭 self.timetable_table = QTableWidget() self.tabs.addTab(self.timetable_table, "시간표") # 급식 탭 self.meal_table = QTableWidget() self.tabs.addTab(self.meal_table, "오늘의 급식") main_layout.addWidget(self.tabs) self.setLayout(main_layout) # 학교 정보를 저장할 변수 self.atpt_ofcdc_sc_code = "" self.sd_sched_dep_code = "" def search_school(self): school_name = self.school_input.text().strip() if not school_name: QMessageBox.warning(self, "경고", "학교명을 입력해주세요.") return url = "https://open.neis.go.kr/hub/schoolInfo" params = { 'KEY': self.api_key, 'Type': 'json', 'pIndex': 1, 'pSize': 5, 'SCHUL_NM': school_name } try: res = requests.get(url, params=params, timeout=5) data = res.json() if 'schoolInfo' in data: info = data['schoolInfo'][1]['row'][0] self.atpt_ofcdc_sc_code = info['ATPT_OFCDC_SC_CODE'] # 시도교육청코드 self.sd_sched_dep_code = info['SD_SCHUL_CODE'] # 행정표준코드 school_full_name = info['SCHUL_NM'] self.info_label.setText(f"선택된 학교: {school_full_name}") # 데이터 불러오기 self.load_timetable() self.load_meal() else: QMessageBox.warning(self, "알림", "검색된 학교가 없습니다.") except Exception as e: QMessageBox.critical(self, "오류", f"학교 정보를 불러오는 중 오류가 발생했습니다:\n{e}") def load_timetable(self): # 초등학교/중학교/고등학교 API 엔드포인트 자동 분기 처리 가능 (기본: 초등/중등) url = "https://open.neis.go.kr/hub/hisTimetable" # 고등: hisTimetable / 중등: misTimetable / 초등: elsTimetable params = { 'KEY': self.api_key, 'Type': 'json', 'ATPT_OFCDC_SC_CODE': self.atpt_ofcdc_sc_code, 'SD_SCHUL_CODE': self.sd_sched_dep_code, 'pIndex': 1, 'pSize': 20 } try: res = requests.get(url, params=params, timeout=5) data = res.json() if 'hisTimetable' in data: rows = data['hisTimetable'][1]['row'] self.timetable_table.setRowCount(len(rows)) self.timetable_table.setColumnCount(3) self.timetable_table.setHorizontalHeaderLabels(["교시", "과목", "일자"]) for idx, row in enumerate(rows): self.timetable_table.setItem(idx, 0, QTableWidgetItem(str(row.get('PERIO', '')))) self.timetable_table.setItem(idx, 1, QTableWidgetItem(str(row.get('ITRT_CNTNT', '')))) self.timetable_table.setItem(idx, 2, QTableWidgetItem(str(row.get('ALL_TI_YMD', '')))) else: self.timetable_table.clear() self.timetable_table.setRowCount(0) except Exception as e: print(f"시간표 로드 오류: {e}") def load_meal(self): url = "https://open.neis.go.kr/hub/mealServiceDietInfo" params = { 'KEY': self.api_key, 'Type': 'json', 'ATPT_OFCDC_SC_CODE': self.atpt_ofcdc_sc_code, 'SD_SCHUL_CODE': self.sd_sched_dep_code, 'pIndex': 1, 'pSize': 5 } try: res = requests.get(url, params=params, timeout=5) data = res.json() if 'mealServiceDietInfo' in data: rows = data['mealServiceDietInfo'][1]['row'] self.meal_table.setRowCount(len(rows)) self.meal_table.setColumnCount(2) self.meal_table.setHorizontalHeaderLabels(["날짜", "메뉴"]) for idx, row in enumerate(rows): # 식단 메뉴 내 HTML 태그 및 공백 정제 dish = row.get('DDISH_NM', '').replace('
', '\n') self.meal_table.setItem(idx, 0, QTableWidgetItem(str(row.get('MLSV_YMD', '')))) self.meal_table.setItem(idx, 1, QTableWidgetItem(dish)) else: self.meal_table.clear() self.meal_table.setRowCount(0) except Exception as e: print(f"급식 로드 오류: {e}") if __name__ == '__main__': app = QApplication(sys.argv) ex = OnSchoolApp() ex.show() sys.exit(app.exec_())
